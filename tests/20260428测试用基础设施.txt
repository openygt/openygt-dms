  # 1. 启动基础设施（已 systemd 化，开机自动起）
  sudo systemctl start docker-stack-mysql-8 docker-stack-emqx docker-stack-nginx                                                                      
                                                                                                                                                      
  # 2. 跑后端                                                                                                                                         
  cd /data2/docker/decoction/openygt-dms-01                                                                                                           
  mvn -pl openygt-dms/dms-app -am spring-boot:run

  # 3. 跑模拟器（QA 这边）
  python3 simulator.py --broker tcp://<server>:1883 --device D001
  python3 simulator.py --broker tcp://<server>:1883 --device D002    # 多开模拟设备群                                                                 
                                                                                                                                                      
  # 4. 浏览器打开                                                                                                                                     
  http://<server>:8080/dashboard-v2.html                                                                                                              
  http://<server>:18083  # EMQX Dashboard，看消息流
